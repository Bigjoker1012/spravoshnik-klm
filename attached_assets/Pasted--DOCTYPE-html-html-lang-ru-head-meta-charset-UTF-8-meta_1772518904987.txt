    <div id="mainGrid" class="p-6 grid grid-cols-2 gap-y-8 justify-items-center">
        <div onclick="filterByDept('Администрация')" class="hexagon-wrapper"><div class="hexagon bg-admin">Администрация</div></div>
        <div onclick="filterByDept('ВЭД')" class="hexagon-wrapper"><div class="hexagon bg-ved">ВЭД</div></div>

        <div onclick="filterByDept('Кадры')" class="hexagon-wrapper"><div class="hexagon bg-hr">Кадры</div></div>
        <div onclick="filterByDept('Финансы')" class="hexagon-wrapper"><div class="hexagon bg-fin">Финансы</div></div>
        <div onclick="filterByDept('Хоз. служба')" class="hexagon-wrapper"><div class="hexagon bg-hoz">Хоз. служба</div></div>
    </div>

    <div id="resetBtn" class="hidden text-center mb-4">
        <button onclick="showAll()" class="text-blue-600 font-medium underline">← Назад к сотам</button>
    </div>

    <div id="employeeList" class="px-4 space-y-4 mt-4">
        </div>

    <script>
        const SHEET_URL = 'https://docs.google.com/spreadsheets/d/1nualyTma75WZ4eZlVuPEPMDqz94qmCx5blby-9tZCOU/export?format=csv';
        let employees = [];

        // 1. Грузим данные
        async function loadData() {
            const response = await fetch(SHEET_URL);
            const data = await response.text();
            const rows = data.split('\n').slice(1); // убираем шапку
            
            employees = rows.map(row => {
                const cols = row.split(','); // ВАЖНО: если в ФИО есть запятые, нужно будет усложнить парсер
                return {
                    name: cols[1]?.trim(),
                    pos: cols[2]?.trim(),
                    dept: cols[3]?.trim(),
                    phone: cols[5]?.trim(),
                    email: cols[6]?.trim(),
                    photo: cols[8]?.trim() || 'https://via.placeholder.com/100'
                };
            }).filter(e => e.name);
            renderEmployees(employees);
        }

        // 2. Рисуем карточки (с твоими любимыми кнопками!)
        function renderEmployees(list) {
            const container = document.getElementById('employeeList');
            container.innerHTML = list.map(e => `
                <div class="bg-white p-4 rounded-2xl shadow-sm flex items-center gap-4 border border-gray-100">
                    <img src="${e.photo}" class="w-16 h-16 rounded-full object-cover bg-gray-200">
                    <div class="flex-1">
                        <div class="font-bold text-gray-800 leading-tight">${e.name}</div>
                        <div class="text-sm text-gray-500 mb-2">${e.pos}</div>
                        <div class="flex gap-3">
                            <a href="tel:${e.phone}" class="text-blue-600 bg-blue-50 p-2 rounded-full">📞</a>
                            <a href="https://wa.me/${e.phone?.replace(/\D/g,'')}" class="text-green-600 bg-green-50 p-2 rounded-full">💬</a>
                            <a href="https://t.me/${e.phone}" class="text-sky-500 bg-sky-50 p-2 rounded-full">✈️</a>
                            <a href="mailto:${e.email}" class="text-gray-500 bg-gray-50 p-2 rounded-full">✉️</a>
                        </div>
                    </div>
                </div>
            `).join('');
        }

        // 3. Фильтры
        function filterByDept(dept) {
            document.getElementById('mainGrid').classList.add('hidden');
            document.getElementById('resetBtn').classList.remove('hidden');
            const filtered = employees.filter(e => e.dept === dept);
            renderEmployees(filtered);
        }

        function showAll() {
            document.getElementById('mainGrid').classList.remove('hidden');
            document.getElementById('resetBtn').classList.add('hidden');
            renderEmployees(employees);
        }

        // Поиск
        document.getElementById('searchInput').addEventListener('input', (e) => {
            const val = e.target.value.toLowerCase();
            if(val) {
                document.getElementById('mainGrid').classList.add('hidden');
                document.getElementById('resetBtn').classList.remove('hidden');
            }
            const filtered = employees.filter(emp => 
                emp.name.toLowerCase().includes(val) || emp.pos.toLowerCase().includes(val)
            );
            renderEmployees(filtered);
        });

        loadData();
    </script>
